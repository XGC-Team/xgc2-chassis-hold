cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(InstalledChassisHoldTests LANGUAGES CXX)
  include(CTest)
  find_package(XgcChassisHold 0.1.0 CONFIG REQUIRED COMPONENTS State Rpc Server)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(HoldJsonCpp REQUIRED IMPORTED_TARGET jsoncpp>=1.7.4)
  find_package(Threads REQUIRED)
endif()
add_executable(chassis_hold_native_test native_test.cpp)
target_link_libraries(chassis_hold_native_test PRIVATE XgcChassisHold::Server XgcChassisHold::Rpc PkgConfig::HoldJsonCpp Threads::Threads)
target_compile_options(chassis_hold_native_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME chassis_hold_native COMMAND chassis_hold_native_test)
set_tests_properties(chassis_hold_native PROPERTIES TIMEOUT 20)
add_executable(chassis_hold_state_test state_test.cpp)
target_link_libraries(chassis_hold_state_test PRIVATE XgcChassisHold::State Threads::Threads)
target_compile_options(chassis_hold_state_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME chassis_hold_state COMMAND chassis_hold_state_test)
add_executable(chassis_hold_cxx17_consumer cxx17_consumer.cpp)
set_target_properties(chassis_hold_cxx17_consumer PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
target_compile_options(chassis_hold_cxx17_consumer PRIVATE -UNDEBUG)
target_link_libraries(chassis_hold_cxx17_consumer PRIVATE XgcChassisHold::Server)
add_test(NAME chassis_hold_cxx17_consumer COMMAND chassis_hold_cxx17_consumer)
